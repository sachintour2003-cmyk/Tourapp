    reviewForm.classList.remove("open");
  });

  /* ------- Submit review ------- */
  reviewForm.addEventListener("submit", async function(e){
    e.preventDefault();

    var name = document.getElementById("rvName").value.trim();
    var text = document.getElementById("rvText").value.trim();
    var btn  = document.getElementById("reviewSubmitBtn");

    if (!name){ showToast("Please enter your name"); return; }
    if (!pickedRating){ showToast("Please select a star rating"); return; }
    if (text.length < 10){ showToast("Please write a few words about your ride"); return; }

    btn.disabled = true;
    btn.textContent = "Submitting...";

    var entry = {
      name:name, rating:pickedRating, text:text,
      date:new Date().toISOString().slice(0, 10)
    };

    var saved = false;

    if (apiReady()){
      try {
        var res = await fetch(API_URL, {
          method:"POST",
          headers:{ "Content-Type":"text/plain;charset=utf-8" },
          body: JSON.stringify({
            action:"create_review",
            name:name, rating:pickedRating, text:text, source:"Website"
          })
        });
        var data = await res.json();
        saved = !!(data && data.ok);
      } catch (err){ saved = false; }
    }

    if (saved){
      allReviews.unshift(entry);
      showToast("Thank you! Your review is now live.", 3000);
    } else {
      entry.pending = true;
      saveLocalReview(entry);
      allReviews.unshift(entry);
      showToast("Thank you! Your review will appear after approval.", 3200);
    }

    renderReviews();
    reviewForm.reset();
    pickedRating = 0;
    paintStars(0);
    reviewForm.classList.remove("open");

    btn.disabled = false;
    btn.textContent = "Submit Review";
  });

  /* ------- Init ------- */
  loadStats();
  initReviews();
})();
